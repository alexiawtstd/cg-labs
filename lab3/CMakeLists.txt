cmake_minimum_required(VERSION 3.20)
include(FetchContent)

project(lab3 LANGUAGES C CXX)

add_executable(${PROJECT_NAME}
	source/main.cpp
	source/graphics_internal.cpp
	source/application.cpp
	source/vk_utils.cpp   # лаб. 3: буферы в видеопамяти, одноразовые команды gpu
	source/texture.cpp    # лаб. 3: загрузка текстур, мип-уровни, сэмплер
)

# SHADER_DIR / TEXTURE_DIR: абсолютные пути к папкам shaders/ и textures/ (подставляет CMake),
# поэтому программа находит файлы, из какой бы папки её ни запускали
target_compile_definitions(${PROJECT_NAME} PRIVATE
	SHADER_DIR="${CMAKE_SOURCE_DIR}/shaders/"
	TEXTURE_DIR="${CMAKE_SOURCE_DIR}/textures/"
)

if(LINUX)
	target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
elseif(APPLE)
	target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
elseif(WIN32)
	if(MSVC)
		target_compile_options(${PROJECT_NAME} PRIVATE /W4 /wd4201 /utf-8)
		target_link_options(${PROJECT_NAME} PRIVATE /subsystem:windows /entry:mainCRTStartup)
	else()
		target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
		target_link_options(${PROJECT_NAME} PRIVATE -mwindows)
	endif()
	target_compile_definitions(${PROJECT_NAME} PRIVATE -D_USE_MATH_DEFINES)
else()
	message(FATAL_ERROR "${CMAKE_SYSTEM_NAME} is not supported")
endif()

set_target_properties(${PROJECT_NAME} PROPERTIES CXX_STANDARD_REQUIRED TRUE CXX_STANDARD 20)

FetchContent_Declare(

)

FetchContent_Declare(

	GIT_REPOSITORY https://github.com/ocornut/imgui.git

)

target_link_libraries(${PROJECT_NAME} PRIVATE

# компиляция шейдеров
# glslc берём из vulkan sdk (путь из CMakePresets.json), иначе ищем в PATH
if(Vulkan_GLSLC_EXECUTABLE)
	set(GLSLC_FOUND ${Vulkan_GLSLC_EXECUTABLE})
else()
	find_program(GLSLC_FOUND glslc)
endif()
if(GLSLC_FOUND)
	set(_SHADER_BINARIES)

	macro(compile_shader SHADER_FILE)
		set(SHADER_SOURCE ${CMAKE_SOURCE_DIR}/shaders/${SHADER_FILE})
		set(SHADER_BINARY ${SHADER_FILE}.spv)
		set(SHADER_BINARY_PATH ${CMAKE_SOURCE_DIR}/shaders/${SHADER_BINARY})

		add_custom_command(
			OUTPUT ${SHADER_BINARY_PATH}
			COMMAND ${GLSLC_FOUND} ${SHADER_SOURCE} -o ${SHADER_BINARY_PATH}
			DEPENDS ${SHADER_SOURCE}
			COMMENT "Compiling ${SHADER_FILE} shader"
		)

		list(APPEND _SHADER_BINARIES ${SHADER_BINARY_PATH})
	endmacro()

	# шейдеры лабораторной работы 3
	compile_shader(lit.vert)
	compile_shader(lit.frag)
	compile_shader(gizmo.vert)
	compile_shader(gizmo.frag)

	add_custom_target(shaders DEPENDS ${_SHADER_BINARIES})
	add_dependencies(${PROJECT_NAME} shaders)
endif()
